  const ensure=(x,m)=>{if(!x)throw new Error(m);report.checks.push(m);};
  const switcher=page.getByRole('button',{name:'气测异常智能解释：气峰背后的工况与迟到',exact:true});
  if(await switcher.count())await switcher.click();
  await page.getByRole('button',{name:'01 专业背景',exact:true}).click();
  ensure(await page.locator('main svg').count()>=3,'Original portal background renders >=3 inline SVGs');

  await page.getByRole('button',{name:'02 案例流程',exact:true}).click();

  const snapshot=()=>f.locator('body').evaluate(()=>M2Inspect.snapshot());
  await f.locator('#run').click();

  await f.locator('#show-truth').check();
  ensure(await f.locator('#metrics-panel').isVisible(),'Gold reveal displays actual metrics');
  await f.locator('.workspace').first().screenshot({path:'G:/可视化导论/5/output/playwright/05-default-curves-sandbox.png'});

  const ev=await f.locator('body').evaluate(()=>M2Inspect.evidence());
  ensure(ev.claims[0].reviewStatus==='rejected'&&ev.claims[0].supportStatus==='conflict','Contradicting manual review rejects model claim instead of silently confirming it');

  ensure(report.download.failure===null,'Evidence JSON downloads inside original sandbox');
  await f.locator('details.controls summary').click();
  await f.locator('#volume').fill('20');
  let s=await snapshot();

  await f.locator('#run').click();

  await f.locator('#volume').fill('0');await f.locator('#run').click();
  ensure((await snapshot()).status==='error','Out-of-range input reports a real error');
  await f.locator('#reset').click();await f.locator('#status').filter({hasText:'计算完成'}).waitFor();

  s=await snapshot();ensure(s.metrics===null&&s.results[0].prediction.label===1,'Same gas with changed operation changes prediction and suppresses gold metrics');
