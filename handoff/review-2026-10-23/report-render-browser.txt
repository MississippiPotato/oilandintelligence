async page => {
  const checks=[];
  const errors=[];
  page.on('pageerror', e=>errors.push(e.message));
  const check=(name,ok,detail)=>{checks.push({name,passed:!!ok,detail});if(!ok)throw new Error(name+': '+JSON.stringify(detail));};
  await page.setViewportSize({width:1440,height:1000});
  check('eleven findings and eleven teacher criteria',await page.locator('.finding').count()===11&&await page.locator('#criteria tbody tr').count()===11,{findings:await page.locator('.finding').count(),teacherRows:await page.locator('#criteria tbody tr').count()});
  await page.getByRole('button',{name:'优先修正',exact:true}).click();
  check('priority filter shows four actionable findings',await page.locator('.finding:not(.hidden)').count()===4,await page.locator('#filter-count').textContent());
  await page.getByRole('button',{name:'教学打磨',exact:true}).click();
  check('teaching filter shows five improvements',await page.locator('.finding:not(.hidden)').count()===5,await page.locator('#filter-count').textContent());
  await page.getByRole('button',{name:'全部',exact:true}).click();
  await page.locator('.gallery img').first().click();
  check('screenshot zoom opens',await page.locator('#zoom').isVisible(),await page.locator('#zoom img').getAttribute('alt'));
  await page.locator('#zoom-close').click();
  check('screenshot zoom closes',!(await page.locator('#zoom').isVisible()),null);
  await page.evaluate(()=>window.scrollTo(0,0));
  await page.screenshot({path:'output/playwright/review-2026-10-23/review-report-desktop.png'});
  check('desktop page fits viewport',await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth),await page.evaluate(()=>({viewport:innerWidth,width:document.documentElement.scrollWidth})));
  await page.setViewportSize({width:390,height:844});
  await page.evaluate(()=>window.scrollTo(0,0));
  await page.screenshot({path:'output/playwright/review-2026-10-23/review-report-mobile.png'});
  check('mobile page fits viewport',await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth),await page.evaluate(()=>({viewport:innerWidth,width:document.documentElement.scrollWidth})));
  const net=[];
  const listener=req=>{if(/^https?:/.test(req.url()))net.push(req.url());};
  page.on('request',listener);
  await page.context().setOffline(true);
  await page.goto('file:///G:/%E5%8F%AF%E8%A7%86%E5%8C%96%E5%AF%BC%E8%AE%BA/5/handoff/review-2026-10-23/index.html');
  for(const im of await page.locator('.gallery img').all())await im.scrollIntoViewIfNeeded();
  const images=await page.locator('.gallery img').evaluateAll(xs=>xs.map(x=>({inline:x.src.startsWith('data:image/'),loaded:x.complete&&x.naturalWidth>0})));
  check('offline file page embeds three loaded screenshots',images.length===3&&images.every(x=>x.inline&&x.loaded),images);
  await page.getByRole('button',{name:'展示便利',exact:true}).click();
  check('offline interaction works',await page.locator('.finding:not(.hidden)').count()===2,await page.locator('#filter-count').textContent());
  check('offline report makes no HTTP requests',net.length===0,net);
  check('report has no browser JavaScript exceptions',errors.length===0,errors);
  await page.context().setOffline(false);
  page.off('request',listener);
  return {checkedAt:new Date().toISOString(),scope:'Actual rendering and interaction of the review artifact, not additional product acceptance',checks,passed:checks.filter(x=>x.passed).length,total:checks.length,errors};
}
