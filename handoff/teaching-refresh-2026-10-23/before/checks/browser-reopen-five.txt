async(page)=>{
const r={scope:'closed and reopened same persistent Chromium profile, five original-portal file packages',checks:[],cases:[],errors:[],httpRequests:[]},ids=['s17-cuttings-lithology','s17-mudgas-context','s17-fluorescence-review','s17-drilltime-boundaries','s17-evidence-report'];
page.on('pageerror',e=>r.errors.push(e.message));page.on('request',q=>{if(/^https?:/.test(q.url()))r.httpRequests.push(q.url());});await page.context().setOffline(true);
for(let i=0;i<ids.length;i++){await page.goto('file:///G:/可视化导论/5/portal-original/index.html#/scene/m'+(i+1)+'?case='+ids[i]+'&tab=demo');await page.reload();const f=page.frameLocator('iframe');await f.locator('#run').click();await f.locator('#status').filter({hasText:'计算完成'}).waitFor();const s=await f.locator('body').evaluate(()=>globalThis.S17Inspect?.snapshot()??M2Inspect.snapshot());r.checks.push({name:'m'+(i+1)+' reopened and computed',status:s.status==='ready'?'passed':'failed'});r.cases.push({id:ids[i],metrics:s.run?.metrics??s.metrics});}
r.persisted=await page.evaluate(async()=>{const records=JSON.parse(localStorage.getItem('atlas.casepack.v2'));return Promise.all(records.map(async r=>({id:r.meta.id,meta:r.meta,pkg:r.pkg,files:await Promise.all(Object.entries(r.files).map(async([name,f])=>{const bytes=new TextEncoder().encode(f.data);return{name:f.name,enc:f.enc,bytes:bytes.length,sha256:Array.from(new Uint8Array(await crypto.subtle.digest('SHA-256',bytes))).map(b=>b.toString(16).padStart(2,'0')).join('')};}))})));});
r.checks.push({name:'all five records persisted',status:r.persisted.length===5?'passed':'failed'});r.checks.push({name:'no errors or HTTP',status:r.errors.length===0&&r.httpRequests.length===0?'passed':'failed'});r.passed=r.checks.filter(c=>c.status==='passed').length;r.failed=r.checks.filter(c=>c.status==='failed').length;return r;
}
